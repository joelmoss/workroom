## 2. Key Technical Concepts

- **libghostty / GhosttyKit**: the embedded terminal engine (Zig-based `ghostty` core), pinned via `libghostty-spm` package (Lakr233), version 1.3.2 / engine commit `35e1a0160c4f6797e1bb1ef8e7a2b8c6b114ab58`.
- **`ghostty_surface_t`**: the C handle for one terminal surface; `GhosttySurfaceView` (`NSView` subclass) wraps it 1:1.
- **`CAMetalLayer` compositing**: libghostty installs its own Metal layer internally on the `NSView` it's handed (`ghostty_platform_macos_s(nsview:)`); Workroom's own Swift never touches `CAMetalLayer` directly.
- **`CATransform3DMakeScale` / `sublayerTransform`**: the mechanism behind "Candidate A" — a compositing-only visual shrink of a surface's rendered output via an ancestor layer's `sublayerTransform`, without touching the surface's actual `.frame.size`.
- **`ghostty_surface_set_size` / `updateMetalLayerSize()`**: the resize path that must NEVER be triggered by a preview feature (confirmed: any `.frame.size` write on a mounted `GhosttySurfaceView` retriggers this via `viewDidMoveToWindow`).
- **`ghostty_surface_set_occlusion` / `GhosttySurfaceView.setVisible(_:)` / `applyOcclusionState()`**: the per-surface GPU-render pause mechanism; `TerminalSessions.reconcileOcclusion(for:)` is documented as the SOLE caller of `setVisible`.
- **`TerminalContainerView.mount(in:)`**: the real production re-homing code for the ACTIVE tab; sets `view.wantsFocus = isFocusedPane` and re-parents the surface via frame+autoresizingMask (never AutoLayout, to avoid stale-constraint bugs from issue #3).
- **`ScreenCaptureKit` / `SCStream` vs `SCScreenshotManager`**: continuous capture vs one-shot; `SCShareableContent.currentProcess` (macOS 14.4+) grants own-process capture with NO Screen Recording permission — empirically confirmed via a standalone compiled Swift binary (not assumed).
- **XCTest / XCUITest**: `@testable import Workroom`; `XCTSkipUnless(GhosttyApp.shared.app != nil, ...)` is the established pattern for tests needing a real libghostty runtime; `.staticTexts.matching(NSPredicate(format: "identifier BEGINSWITH %@", ...))` is the established pattern for reliably counting/indexing tab chips (vs `.descendants(matching: .any)`, which double-matches title+close-button).
- **Claude Code workflow tools used**: `EnterPlanMode`/`ExitPlanMode` (native plan mode), `/plan-eng-review` (gstack skill — heavy structured review with AskUserQuestion gates, Codex outside-voice via `codex exec`, mandatory `## GSTACK REVIEW REPORT` footer with exact unbolded `NO UNRESOLVED DECISIONS` sentinel), `Agent` (Explore/Plan subagents), `AskUserQuestion`.
- **Sandbox/build conventions**: `dangerouslyDisableSandbox: true` required for `make app-*`, `screencapture`, `osascript` real-mouse driving; `make app-generate` (xcodegen) required after adding/removing Swift files before direct `xcodebuild -only-testing` picks them up.
- **Session-tokened claims**: `PreviewClaim { tabID, session }` pattern to prevent a stale/superseded async session's teardown from clobbering a newer session's still-active state — a generation counter used as both cancellation token AND occlusion-claim session identity.
- **Dependency-injection testability seam**: `PreviewScheduling`/`PreviewCancellable` (wraps `DispatchQueue.main.asyncAfter`) and `PreviewFrameSource` protocol (real: `TransformScaleFrameSource`; fake: `FakeFrameSource` in tests) — added specifically to keep the state machine unit-testable without resurrecting a full swappable-capture-backend abstraction (a scope cut, "D3", made during eng review).

```swift
  previewClaims[target.id] = PreviewClaim(tabID: tabID, session: session)
  reconcileOcclusion(for: target)
}

func endPreview(_ tabID: TerminalTab.ID, session: Int, for target: TerminalTarget) {
  guard previewClaims[target.id]?.session == session else { return }
  previewClaims[target.id] = nil
  reconcileOcclusion(for: target)

func reconcileOcclusion(for target: TerminalTarget) {
  var visible = Set(visibleTabIDs(for: target))
  if let claim = previewClaims[target.id] { visible.insert(claim.tabID) }
  for tab in (tabsByTarget[target.id] ?? [:]).values {
    tab.surface?.setVisible(visible.contains(tab.id))
  }
}

func isPreviewEligible(_ tab: TerminalTab, for target: TerminalTarget) -> Bool {
  tab.surface?.hasSurface == true && !visibleTabIDs(for: target).contains(tab.id)
}
```

```swift
protocol PreviewCancellable { func cancel() }
protocol PreviewScheduling {
  @discardableResult
  func schedule(after delay: TimeInterval, _ block: @escaping () -> Void) -> PreviewCancellable
}
final class DispatchQueueScheduling: PreviewScheduling { ... }

protocol PreviewFrameSource: AnyObject {
  func mount(surface: GhosttySurfaceView, host: NSView) -> Bool
  func startPresenting(onFirstFrame: @escaping () -> Void)
  func stopPresenting()
  func unmount()
}

final class TransformScaleFrameSource: PreviewFrameSource {
  private let scheduling: PreviewScheduling
  private weak var mountedSurface: GhosttySurfaceView?
  private var settleWork: PreviewCancellable?

  func mount(surface: GhosttySurfaceView, host: NSView) -> Bool {
    let originalSize = surface.frame.size
    guard originalSize.width > 0, originalSize.height > 0 else { return false }
    let hostSize = host.bounds.size
    guard hostSize.width > 0, hostSize.height > 0 else { return false }
    let scale = min(hostSize.width / originalSize.width, hostSize.height / originalSize.height)
    guard scale.isFinite, scale > 0 else { return false }
    host.wantsLayer = true
    host.layer?.masksToBounds = true
    let origin = NSPoint(
      x: hostSize.width / 2 - originalSize.width / 2,
      y: hostSize.height / 2 - originalSize.height / 2)
    surface.frame = NSRect(origin: origin, size: originalSize)
    surface.autoresizingMask = []
    // FIX (bug #2, wantsFocus): force false before ever re-entering a window.
    surface.wantsFocus = false
    host.addSubview(surface)
    host.layer?.sublayerTransform = CATransform3DMakeScale(scale, scale, 1)
    mountedSurface = surface
    return true
  }
  func startPresenting(onFirstFrame: @escaping () -> Void) {
    settleWork = scheduling.schedule(after: 0.05, onFirstFrame)
  }
  func stopPresenting() { settleWork?.cancel(); settleWork = nil }
  func unmount() { mountedSurface?.removeFromSuperview(); mountedSurface = nil }
}

final class PreviewHostView: NSView {
  override func hitTest(_ point: NSPoint) -> NSView? { nil }
}

@MainActor
final class TerminalHoverPreviewController: ObservableObject {
  enum Phase: Equatable { case idle; case armed(tabID: TerminalTab.ID); case visible(tabID: TerminalTab.ID) }
  @Published private(set) var phase: Phase = .idle
  @Published private(set) var previewHost: NSView?
  private(set) var generation = 0
  private var activeSession: Int?  // FIX (bug: generation-vs-session-token) — NOT the live `generation`
  ...

  func armHover(tab: TerminalTab, target: TerminalTarget) {
    guard sessions.isPreviewEligible(tab, for: target) else { return }
    cancelHover()
    let gen = generation
    phase = .armed(tabID: tab.id)
    armWork = scheduling.schedule(after: dwell) { [weak self] in
      self?.resolve(tabID: tab.id, target: target, gen: gen)
    }
  }

  func cancelHover() {
    armWork?.cancel(); armWork = nil; generation += 1
    if case .visible = phase { teardown() } else { phase = .idle }
  }

  private func resolve(tabID: TerminalTab.ID, target: TerminalTarget, gen: Int) {
    guard gen == generation else { return }
    guard let tab = sessions.tab(tabID, for: target) else { phase = .idle; return }
    guard sessions.isPreviewEligible(tab, for: target) else { phase = .idle; return }
    guard let surface = tab.surface else { phase = .idle; return }
    mount(surface: surface, tabID: tabID, target: target, gen: gen)
  }

  private func mount(surface: GhosttySurfaceView, tabID: TerminalTab.ID, target: TerminalTarget, gen: Int) {
    // FIX (bug #1, superview guard): superview is now a PLAIN OPTIONAL, not force-unwrapped —
    // a genuinely backgrounded tab's surface has ALREADY been detached (superview == nil), which
    // is the entire premise of why re-homing is needed. The old `guard let superview = ... else
    // { return }` silently no-op'd on every real hover.
    let superview = surface.superview
    let preMountFrame = surface.frame
    let host = PreviewHostView(frame: NSRect(x: 0, y: 0, width: 260, height: 160))
    let frameSource = makeFrameSource()
    guard frameSource.mount(surface: surface, host: host) else { phase = .idle; return }
    activeFrameSource = frameSource
    mountedSurface = surface
    originalSuperview = superview
    originalFrame = preMountFrame
    activeTarget = target; activeTabID = tabID; activeSession = gen
    sessions.beginPreview(tabID, session: gen, for: target)
    guard gen == generation else { teardownAllocatedState(); return }
    previewHost = host
    frameSource.startPresenting { [weak self] in
      guard let self, gen == self.generation else { return }
      self.phase = .visible(tabID: tabID)
    }
  }

  private func teardown() { teardownAllocatedState(); phase = .idle }

  private func teardownAllocatedState() {
    activeFrameSource?.stopPresenting(); activeFrameSource?.unmount(); activeFrameSource = nil
    if let surface = mountedSurface, let superview = originalSuperview, let frame = originalFrame {
      surface.frame = frame; surface.autoresizingMask = [.width, .height]; superview.addSubview(surface)
    }
    if let target = activeTarget, let tabID = activeTabID, let session = activeSession {
      sessions.endPreview(tabID, session: session, for: target)
    }
    mountedSurface = nil; originalSuperview = nil; originalFrame = nil; previewHost = nil
    activeTarget = nil; activeTabID = nil; activeSession = nil
  }
}
```

```swift
import XCTest

/// Regression coverage for the hover-preview feature (plan: ~/.claude/plans/ethereal-inventing-wolf.md)
/// stealing keyboard focus from the real active tab — found via real-mouse QA, not by any automated
/// test: hovering a backgrounded tab's chip silently made it the first responder (a stale
/// `GhosttySurfaceView.wantsFocus` left over from when it was last actually active triggered
/// `viewDidMoveToWindow`'s auto-claim once the hover-preview re-homed it into a real window).
///
/// The surface itself is Metal-rendered and exposes nothing queryable for "who currently holds first
/// responder" (the same CMT-3 gap `GhosttyActionDispatchUITests` documents), so this proves the
/// regression the way a real user would notice it: type into the real active tab, hover a DIFFERENT
/// tab's chip, then type again and confirm the keystrokes still land in the ORIGINALLY-FOCUSED tab's
/// chip POSITION — exercised through the shell's own OSC 0/2 retitling (issue #2), the same detector
/// `GhosttyActionDispatchUITests.testRunningCommandRetitlesTheTabChip` already uses. Checked by strip
/// POSITION, not by re-querying for the expected title prefix — a title-prefix query would still find
/// A match even if the WRONG chip (the hovered one) is the one that actually retitled, which would
/// silently pass right through the regression this test exists to catch.
///
/// Run with `make app-uitest` on a real GUI login session (XCUITest can't drive a headless run).
final class TerminalHoverPreviewUITests: XCTestCase {
  override func setUp() {
    super.setUp()
    continueAfterFailure = false
  }

  private func launchedApp() -> XCUIApplication {
    let app = XCUIApplication()
    app.launchArguments += ["-WorkroomUITestFixture", "1"]
    app.launchArguments += ["-WorkroomUITestNoRunCommand", "1"]
    app.launchArguments += ["-WorkroomUITestTerminalTabs", "2"]
    app.launchArguments += ["-ApplePersistenceIgnoreState", "YES"]
    app.launch()
    XCTAssertTrue(app.wait(for: .runningForeground, timeout: 15))
    return app
  }

  /// `.staticTexts` specifically, not `.descendants(matching: .any)` — the chip's title AND its close
  /// button share the same `terminal.tab.<title>` identifier prefix (`TabStripOverflowUITests`' own
  /// comment: "match only the title StaticText to count chips 1:1"), so an `.any` query returns more
  /// than one element per chip and breaks positional indexing.
  private func chipQuery(_ app: XCUIApplication) -> XCUIElementQuery {
    app.staticTexts.matching(NSPredicate(format: "identifier BEGINSWITH %@", "terminal.tab."))
  }

  private func chip(_ app: XCUIApplication, at index: Int) -> XCUIElement {
    chipQuery(app).element(boundBy: index)
  }

  private func focusFirstTab(_ app: XCUIApplication) {
    app.activate()
    let query = chipQuery(app)
    XCTAssertTrue(query.firstMatch.waitForExistence(timeout: 20))
    XCTAssertGreaterThanOrEqual(query.count, 2, "the fixture must seed 2 terminal tabs")
    chip(app, at: 0).click()
    RunLoop.current.run(until: Date().addingTimeInterval(1))
  }

  private func run(_ app: XCUIApplication, _ line: String) {
    app.typeText(line + "\r")
  }

  private func waitForChipLabel(
    _ app: XCUIApplication, at index: Int, toContain substring: String, timeout: TimeInterval = 20
  ) -> Bool {
    let deadline = Date().addingTimeInterval(timeout)
    while Date() < deadline {
      if chip(app, at: index).label.contains(substring) { return true }
      RunLoop.current.run(until: Date().addingTimeInterval(0.25))
    }
    return false
  }

  func testHoveringABackgroundTabDoesNotStealFocusFromTheActiveTab() {
    let app = launchedApp()
    focusFirstTab(app)
    run(app, "sleep 3; echo before_hover_marker")
    XCTAssertTrue(waitForChipLabel(app, at: 0, toContain: "before_hover"), "...")
    chip(app, at: 1).hover()
    RunLoop.current.run(until: Date().addingTimeInterval(1.0))
    run(app, "sleep 3; echo after_hover_marker")
    XCTAssertTrue(waitForChipLabel(app, at: 0, toContain: "after_hover"), "...")
    XCTAssertFalse(chip(app, at: 1).label.contains("after_hover"), "...")
  }

  func testHoveringTheAlreadyVisibleTabIsANoOp() {
    let app = launchedApp()
    focusFirstTab(app)
    chip(app, at: 0).hover()
    RunLoop.current.run(until: Date().addingTimeInterval(1.0))
    run(app, "sleep 3; echo still_focused_marker")
    XCTAssertTrue(waitForChipLabel(app, at: 0, toContain: "still_focused"), "...")
  }
}
```
