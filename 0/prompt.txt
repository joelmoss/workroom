## Architecture summary

A new data-driven registry (`ToolLogoRegistry`) sits alongside — not inside — the existing `AgentBackend` enum. It's populated from a bundled JSON file that is the **single source of truth** shared by a new bash fetch script and the Swift app, so the "what to fetch" list and the "what to match" list can never drift apart. Detection reuses the exact latch/clear lifecycle already proven for `activeAgentBackend` (`TerminalSessions.swift`), and fixes a real correctness bug (`proc_name` vs `argv0`) in a shared low-level helper so both the existing inline-diagnosis feature and the new favicon feature resolve PIDs correctly. Rendering is one new block in `TerminalTabChip`'s existing content-glyph slot, gated by "does this tab have a recognized tool with an actually-bundled logo" — never a fallback glyph.

```swift
import AppKit

/// One curated entry (issue #141): identifies a CLI/TUI by its exec'd basename (or, for tools that
/// rewrite their own title, a literal provider title) and names the bundled `ToolLogo-&lt;id&gt;` imageset
/// carrying its real brand favicon. Loaded from Resources/tool-logos/registry.json — the SAME file
/// Scripts/fetch-tool-logos.sh reads to know what to fetch, so the fetch list and the match list
/// can't drift apart.
struct ToolLogoEntry: Codable, Equatable {
  let id: String
  let displayName: String
  let executables: [String]
  let titles: [String]?       // provider-owned literal titles (e.g. "claude code"); rare
  let faviconSource: String   // consumed only by the fetch script; decoded here too, unused at runtime
}

/// Latched onto a running terminal tab (mirrors `AgentBackend`, but data-driven and not limited to
/// two cases). Deliberately smaller than `ToolLogoEntry` — a tab only needs to remember the match.
struct RecognizedTool: Equatable {
  let id: String
  let displayName: String
}

enum ToolLogoRegistry {
  static let entries: [ToolLogoEntry] = loadEntries()

  private static let byExecutable: [String: ToolLogoEntry] = {
    var map: [String: ToolLogoEntry] = [:]
    for entry in entries { for exe in entry.executables { map[exe] = entry } }
    return map
  }()

  private static let byTitle: [String: ToolLogoEntry] = {
    var map: [String: ToolLogoEntry] = [:]
    for entry in entries { for t in entry.titles ?? [] { map[t.lowercased()] = entry } }
    return map
  }()

  /// Pure data match — works even before `fetch-tool-logos.sh` has produced an imageset for the id,
  /// so it's directly unit-testable against registry.json content alone.
  ///
  /// Case-sensitive on purpose, mirroring `AgentTitleRecognition.backend(for:)`
  /// (AgentUsage.swift:105-126): a command title is the real, lower-case executable name, so
  /// comparing case-sensitively rejects prose like "Go to definition" matching the `go` entry.
  /// DO NOT `.lowercased()` this side — that regression multiplies with ~30 registry entries.
  static func matchingEntry(forTitle title: String?) -&gt; ToolLogoEntry? {
    guard let title else { return nil }
    let trimmed = title.trimmingCharacters(in: .whitespacesAndNewlines)
    guard !trimmed.isEmpty else { return nil }
    if let literal = byTitle[trimmed.lowercased()] { return literal }
    guard let first = trimmed.split(whereSeparator: \.isWhitespace).first else { return nil }
    let token = String(first).trimmingCharacters(in: CharacterSet(charactersIn: "\"'"))
    return byExecutable[(token as NSString).lastPathComponent]
  }

  static func matchingEntry(forExecutableName name: String?) -&gt; ToolLogoEntry? {
    guard let name else { return nil }
    return byExecutable[(name as NSString).lastPathComponent]
  }

  /// The gate production call sites actually use: nil unless the id's real logo made it into
  /// Assets.xcassets. A registry.json entry can exist before the fetch script has been run for it
  /// (or a fetch failed for one tool) — this must never surface a broken/blank image.
  static func tool(forTitle title: String?) -&gt; RecognizedTool? {
    matchingEntry(forTitle: title).flatMap(recognizedIfBundled)
  }
  static func tool(forExecutableName name: String?) -&gt; RecognizedTool? {
    matchingEntry(forExecutableName: name).flatMap(recognizedIfBundled)
  }

  static func assetName(for id: String) -&gt; String { "ToolLogo-\(id)" }

  private static func recognizedIfBundled(_ entry: ToolLogoEntry) -&gt; RecognizedTool? {
    guard NSImage(named: assetName(for: entry.id)) != nil else { return nil }
    return RecognizedTool(id: entry.id, displayName: entry.displayName)
  }

  private static func loadEntries() -&gt; [ToolLogoEntry] {
    guard
      let url = Bundle.main.url(
        forResource: "registry", withExtension: "json", subdirectory: "tool-logos"),
      let data = try? Data(contentsOf: url),
      let decoded = try? JSONDecoder().decode([ToolLogoEntry].self, from: data)
    else { return [] }
    return decoded
  }
}
```

**3. `macapp/WorkroomSessionProtocol/ProcessArgv0.swift`** — the correctness-bug fix, extracted verbatim from `SessionPTY.swift:154-181`:
```swift
import Darwin

/// The basename of `argv[0]` as a process was originally exec'd (e.g. `claude`, `vim`), or nil if it
/// can't be resolved. Deliberately NOT `proc_name`/`p_comm`: some CLIs (Claude Code among them,
/// observed in the wild) rename their own process title after launch — `p_comm` then reads as their
/// version string ("2.1.232"), not the command a user typed. `KERN_PROCARGS2` returns the ORIGINAL
/// exec-time argv snapshot, unaffected by any later self-renaming.
///
/// Shared by both PID-resolution paths in the app (issue #141): the session daemon's reattach-time
/// foreground detection (`SessionDaemon.swift`, historically the only caller) and the live-pane path
/// (`GhosttySurfaceView`'s `foregroundAgentBackend`/`foregroundTool`) — which previously used the
/// buggy `proc_name` and is fixed by switching to this shared implementation.
public enum ProcessArgv0 {
  public static func executableName(processID: pid_t) -&gt; String? {
    guard processID &gt; 0 else { return nil }
    var mib: [Int32] = [CTL_KERN, KERN_PROCARGS2, processID]
    var size = 0
    guard sysctl(&amp;mib, 3, nil, &amp;size, nil, 0) == 0, size &gt; 4 else { return nil }
    var buffer = [UInt8](repeating: 0, count: size)
    guard sysctl(&amp;mib, 3, &amp;buffer, &amp;size, nil, 0) == 0, size &gt; 4 else { return nil }
    let argc = buffer.withUnsafeBytes { $0.loadUnaligned(as: Int32.self) }
    guard argc &gt; 0 else { return nil }
    var offset = 4
    while offset &lt; size, buffer[offset] != 0 { offset += 1 }
    while offset &lt; size, buffer[offset] == 0 { offset += 1 }
    guard offset &lt; size else { return nil }
    let start = offset
    while offset &lt; size, buffer[offset] != 0 { offset += 1 }
    guard offset &gt; start else { return nil }
    let argv0 = String(decoding: buffer[start..&lt;offset], as: UTF8.self)
    guard let lastSlash = argv0.lastIndex(of: "/") else { return argv0 }
    let basename = argv0[argv0.index(after: lastSlash)...]
    return basename.isEmpty ? nil : String(basename)
  }
}
```

## Fetch script (`macapp/Scripts/fetch-tool-logos.sh`)

Bash, following `vendor-markdown.sh`'s exact conventions (`curl` + `python3 -c` for JSON, no new dependency):

```bash
#!/usr/bin/env bash
# Fetch/refresh curated tool-logo assets (issue #141). Reads Resources/tool-logos/registry.json and,
# per entry, writes an Assets.xcassets/ToolLogo-&lt;id&gt;.imageset from its real favicon.
#
#   macapp/Scripts/fetch-tool-logos.sh            # refresh every entry
#   macapp/Scripts/fetch-tool-logos.sh claude gh  # refresh just these ids
#
# Highest quality first: apple-touch-icon.png (typically 180×180) → Google's public favicon proxy at
# 256px (resolves whatever a site actually declares, without this script parsing HTML) → favicon.ico
# as a last resort. ALWAYS eyeball the fetched PNGs once — Google's proxy succeeds with a generic
# globe for sites it can't resolve, so a bad fetch doesn't error, it just looks wrong.
set -euo pipefail
DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." &amp;&amp; pwd)"
REGISTRY="$DIR/Resources/tool-logos/registry.json"
ASSETS="$DIR/WorkroomApp/Assets.xcassets"
TMP="$(mktemp -d)"; trap 'rm -rf "$TMP"' EXIT
ids_filter=("$@")

wanted() { [[ ${#ids_filter[@]} -eq 0 ]] &amp;&amp; return 0; for w in "${ids_filter[@]}"; do [[ "$w" == "$1" ]] &amp;&amp; return 0; done; return 1; }

fetch_source() {
  local id="$1" source="$2" domain
  domain="$(sed -E 's#^https?://##; s#/.*##' &lt;&lt;&lt;"$source")"
  for url in "$source/apple-touch-icon.png" \
             "https://www.google.com/s2/favicons?domain=$domain&amp;sz=256" \
             "$source/favicon.ico"; do
    if curl -fsSL --max-time 10 "$url" -o "$TMP/$id.raw" 2&gt;/dev/null \
      &amp;&amp; [[ -s "$TMP/$id.raw" ]] \
      &amp;&amp; sips -s format png "$TMP/$id.raw" --out "$TMP/$id.png" &gt;/dev/null 2&gt;&amp;1
    then echo "  $id ← $url"; return 0; fi
  done
  return 1
}

write_imageset() {
  local id="$1" dir="$ASSETS/ToolLogo-$id.imageset"
  mkdir -p "$dir"
  sips -Z 32 "$TMP/$id.png" --out "$dir/logo_32.png" &gt;/dev/null   # -Z: proportional, not force-square
  sips -Z 64 "$TMP/$id.png" --out "$dir/logo_32@2x.png" &gt;/dev/null
  cat &gt;"$dir/Contents.json" &lt;&lt;JSON
{
  "images" : [
    { "idiom" : "universal", "scale" : "1x", "filename" : "logo_32.png" },
    { "idiom" : "universal", "scale" : "2x", "filename" : "logo_32@2x.png" }
  ],
  "info" : { "author" : "xcode", "version" : 1 }
}
JSON
}

failed=()
while IFS=$'\t' read -r id source; do
  wanted "$id" || continue
  if fetch_source "$id" "$source"; then write_imageset "$id"
  else echo "  $id: no favicon fetched" &gt;&amp;2; failed+=("$id"); fi
done &lt; &lt;(python3 -c '
import json, sys
for e in json.load(open(sys.argv[1])): print(f"{e[\"id\"]}\t{e[\"faviconSource\"]}")
' "$REGISTRY")

echo "Done. Eyeball the ~30 PNGs under $ASSETS/ToolLogo-*.imageset before committing —"
echo "Google's proxy succeeds silently with a generic globe for unresolvable domains."
[[ ${#failed[@]} -gt 0 ]] &amp;&amp; { echo "warning: no logo for: ${failed[*]}" &gt;&amp;2; exit 1; }
```

No VENDOR.md-style checksum manifest (unlike `vendor-markdown.sh`) — that exists there because DOMPurify is security-critical (sanitizes untrusted content); these are decorative brand icons with no such risk, so a checksum ledger would be unrequested complexity.
