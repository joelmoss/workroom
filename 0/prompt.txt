+/// The theme picker's window (issue #36) — a floating, movable utility panel.
+///
+/// **Why this isn't the `.sheet` it used to be.** Choosing a theme is a live-preview gesture: ↑/↓
+/// apply families as you move through them, so the app *behind* the picker is the preview. A sheet
+/// defeated that twice over, both measured on the real build:
+///
+/// - **It faded the app.** A sheet is document-modal, so AppKit disables and inactive-renders the
+///   window it's attached to. Side-by-side captures of the same window: traffic lights greyed out,
+///   the sidebar's VCS dot badges went from orange to a dull brown, and the detail placeholder
+///   washed out. You were judging a theme against colours the app never shows while you use it.
+/// - **It couldn't be moved.** A sheet is anchored under its window's title bar, so it sat on top of
+///   the content whose colours you were trying to see, with nothing to drag.
+///
+/// An `NSPanel` fixes both: `NSPanel` refuses to become the *main* window, so the workroom window
+/// keeps `isMainWindow` and goes on drawing active while this panel holds the keyboard, and a titled
+/// panel drags anywhere on screen — including clear of the window entirely.
+///
+/// `SwitcherPanel` is the app's other panel and the mirror image of this one: it must *never* take
+/// key focus (the ⌥Tab gesture depends on the workroom window keeping it), whereas this one must,
+/// because the search field and the live-apply ↑/↓ keys are the whole interaction.
+final class ThemePanel: NSPanel {
+  /// Esc closes, matching the sheet this replaces. `NSWindow` routes `cancelOperation` to a close for
+  /// sheets and modal panels only, so a floating panel has to opt in.
+  override func cancelOperation(_ sender: Any?) { performClose(sender) }
+}

+  /// focused terminal (the two surfaces a theme is judged on) unobstructed.

+      panel.performClose(nil)

+  /// Build the panel, ordered out. Not private: `ThemePanelTests` asserts the **production**
+  /// configuration through this, rather than re-declaring a panel in the test and checking the
+  /// properties it just set itself.
+  func makePanel() -> ThemePanel {
+    let panel = ThemePanel(
+      contentRect: NSRect(origin: .zero, size: Self.contentSize),
+      // No `.resizable`: the content is a fixed-width list, and the picker frames itself.
+      styleMask: [.titled, .closable, .utilityWindow], backing: .buffered, defer: false)
+    panel.title = "Theme"
+    // Floats above the workroom window, so clicking a row (which hands key focus back) can't bury the
+    // panel behind the window whose colours it is changing.

+    // Usable over a full-screen workroom window; without this the panel simply never shows there.
+    panel.collectionBehavior.insert(.fullScreenAuxiliary)

+  }

+    panel.backgroundColor = ThemeService.shared.tokens.nsPanel
+  }

+  /// Restore the user's own placement, else park the panel at the anchor window's top-trailing corner.
+  ///

+  /// attached restores off-screen, and a panel the user cannot see is a panel they cannot close.
+  private func positionIfNeeded(_ panel: ThemePanel) {

+    // `UserDefaults`, which `cfprefsd` resolves against the real home whatever `$HOME` a test run sets.
+    // So a test that drags the panel would move where it opens for the developer, permanently — and a

+    if !UITestFixture.isActive {
+      panel.setFrameAutosaveName(Self.frameAutosaveName)
+      let screens = NSScreen.screens.map(\.visibleFrame)
+      if panel.setFrameUsingName(Self.frameAutosaveName),
+        Self.isOnScreen(panel.frame, screens: screens)
+      {
+        return
+      }
+    }
+    let size = panel.frame.size
+    let origin = Self.defaultOrigin(
+      size: size, anchor: NSApp.mainWindow?.frame, visible: NSScreen.main?.visibleFrame)
+    panel.setFrame(NSRect(origin: origin, size: size), display: false)
+  }
+
+  /// Whether `frame` overlaps a visible area of some attached screen.
+  static func isOnScreen(_ frame: NSRect, screens: [NSRect]) -> Bool {
+    screens.contains { $0.intersects(frame) }
+  }
+
+  /// Top-trailing corner of the window a theme would be judged against — the main window (never this
+  /// panel, which cannot become main), falling back to the screen when no window is open.
+  static func defaultOrigin(size: CGSize, anchor: NSRect?, visible: NSRect?) -> CGPoint {
+    let bounds = anchor ?? visible ?? NSRect(x: 0, y: 0, width: 1440, height: 900)
+    var origin = CGPoint(
+      x: bounds.maxX - size.width - defaultInset, y: bounds.maxY - size.height - defaultInset)
+    // Clamp into the screen: a workroom window can sit partly off the right edge, and the panel
+    // would inherit that.
+    if let visible {
+      origin.x = min(max(origin.x, visible.minX), max(visible.maxX - size.width, visible.minX))
+      origin.y = min(max(origin.y, visible.minY), max(visible.maxY - size.height, visible.minY))
+    }
+    return origin
+  }

+/// `ThemePanel` (issue #36): the window-level invariants that make a *live* theme preview possible.
+///
+/// The picker used to be a `.sheet`, and that broke previewing in two ways a colour test can't see —
+/// the sheet's modality made AppKit inactive-render the window behind it (measured: traffic lights
+/// greyed, VCS dot badges orange → dull brown), and a sheet is anchored, so it covered the very
+/// content whose colours were being chosen. Both symptoms are properties of the *window*, so this is
+/// where they're pinned. Assertions run against the panel `ThemePanelController` actually builds.
+@MainActor
+final class ThemePanelTests: XCTestCase {
+
+  /// Showing the panel makes AppKit persist its frame, and the app host's `UserDefaults` is the
+  /// developer's own Debug-app domain — no `$HOME` isolates it. So put back whatever was there.
+  private var savedFrame: Any?
+
+  override func setUp() {
+    savedFrame = UserDefaults.standard.object(
+      forKey: ThemePanelController.frameAutosaveDefaultsKey)
+  }
+
+  override func tearDown() {
+    let key = ThemePanelController.frameAutosaveDefaultsKey
+    if let savedFrame {
+      UserDefaults.standard.set(savedFrame, forKey: key)
+    } else {
+      UserDefaults.standard.removeObject(forKey: key)
+    }
+  }
+
+  private func makePanel() -> ThemePanel { ThemePanelController().makePanel() }
+
+  func testThePanelCannotBecomeMainSoTheWorkroomWindowKeepsDrawingActive() {
+    // THE fix. `NSPanel` refuses main, so `NSApp.mainWindow` stays the workroom window while this
+    // panel holds the keyboard — which is what stops the app fading out behind the picker. Pinned
+    // because "make it a plain NSWindow" or "go back to a sheet" both look harmless and both
+    // reintroduce the bug.
+    XCTAssertFalse(makePanel().canBecomeMain, "a main-capable window here would fade the app again")
+  }
+
+  func testThePanelTakesKeyFocusSoTheSearchFieldAndArrowsWork() {
+    // The mirror image of `SwitcherPanel`, which must never be key. Here the search field and the
+    // live-applying ↑/↓ are the entire interaction, so key focus is mandatory.
+    let panel = makePanel()
+    XCTAssertTrue(panel.canBecomeKey)
+    XCTAssertFalse(
+      panel.becomesKeyOnlyIfNeeded, "it must take key on open, not wait for a text-field click")
+  }
+
+  func testThePanelIsMovable() {
+    // The second half of the report: a sheet has nothing to drag. A title bar does.
+    let panel = makePanel()
+    XCTAssertTrue(panel.isMovable)
+    XCTAssertTrue(panel.styleMask.contains(.titled), "the title bar is the drag handle")
+    XCTAssertTrue(panel.styleMask.contains(.closable))
+  }
+
+  func testThePanelFloatsAboveTheWindowItRecolours() {
+    // Clicking a row hands key focus back to the workroom window; without floating, the panel would
+    // drop behind the window whose colours it is changing.

+    XCTAssertTrue(panel.isFloatingPanel)
+    XCTAssertTrue(
+      panel.collectionBehavior.contains(.fullScreenAuxiliary),
+      "a panel that can't show over a full-screen window is a picker half the users never see")
+  }
+
+  func testEscapeClosesThePanel() {
+    // `NSWindow` routes `cancelOperation` to a close for sheets and modal panels only, so a floating
+    // panel has to opt in — and Esc is what the sheet answered to.
+    let panel = makePanel()

+  func testAWindowHangingOffTheScreenEdgeDoesNotDragThePanelOffWithIt() {
